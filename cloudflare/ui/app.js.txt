const form = document.querySelector("#settings-form");
const notice = document.querySelector("#notice");
const runError = document.querySelector("#run-error");
const pauseButton = document.querySelector("#pause-button");
const connectButton = document.querySelector("#connect-button");
const connectDialog = document.querySelector("#connect-dialog");
const connectForm = document.querySelector("#connect-form");
const connectError = document.querySelector("#connect-error");
const saveButton = document.querySelector("#save-button");
const levelNames = ["early", "middle", "late"];
let settings = null;
let paused = false;
let currentStatus = { connected: false, configured: false, paused: false, controlEnabled: false };

function setNotice(message, kind = "success") {
  notice.textContent = message;
  notice.className = `notice ${kind}`;
  notice.hidden = !message;
}

async function request(path, options = {}) {
  const response = await fetch(path, {
    ...options,
    headers: { "content-type": "application/json", ...options.headers },
    credentials: "same-origin",
  });
  const payload = response.status === 204 ? {} : await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload.error || `Request failed (${response.status})`);
  return payload;
}

function formatDate(value) {
  if (!value) return "—";
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return value;
  return new Intl.DateTimeFormat(undefined, {
    weekday: "short",
    hour: "numeric",
    minute: "2-digit",
  }).format(date);
}

function formatClock(value, timezone) {
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return "—";
  const showSeconds = date.getSeconds() !== 0;
  try {
    return new Intl.DateTimeFormat(undefined, {
      timeZone: timezone || "UTC",
      hour: "numeric",
      minute: "2-digit",
      ...(showSeconds ? { second: "2-digit" } : {}),
    }).format(date);
  } catch {
    return new Intl.DateTimeFormat(undefined, {
      hour: "numeric",
      minute: "2-digit",
      ...(showSeconds ? { second: "2-digit" } : {}),
    }).format(date);
  }
}

function renderTimeline(status) {
  const timeline = status?.timeline;
  const panel = document.querySelector("#timeline-panel");
  const timeList = document.querySelector("#phase-time-list");
  const timeBox = document.querySelector("#phase-times");
  const chart = document.querySelector("#timeline-chart");
  const nowLabel = document.querySelector("#timeline-now");
  timeList.replaceChildren();
  chart.replaceChildren();
  nowLabel.hidden = true;
  const segments = Array.isArray(timeline?.segments)
    ? timeline.segments
        .map((segment) => ({
          ...segment,
          start: new Date(segment.startAt).getTime(),
          end: new Date(segment.endAt).getTime(),
        }))
        .filter(
          (segment) =>
            Number.isFinite(segment.start) &&
            Number.isFinite(segment.end) &&
            segment.end >= segment.start,
        )
    : [];
  if (!segments.length) {
    panel.hidden = true;
    timeBox.hidden = true;
    return;
  }

  const timezone = timeline.timezone || settings?.timezone || "UTC";
  const phaseOrder = ["preheat", "early", "middle", "late"];
  const phases = phaseOrder
    .map((key) => segments.find((segment) => segment.key === key))
    .filter(Boolean);
  if (phases.length) {
    for (const phase of phases) {
      const row = document.createElement("div");
      row.className = "phase-time-row";
      const name = document.createElement("span");
      name.textContent = phase.label || phase.key;
      const times = document.createElement("strong");
      times.textContent =
        phase.key === "preheat" && phase.start === phase.end
          ? "No gap before bedtime"
          : `${formatClock(phase.startAt, timezone)} – ${formatClock(phase.endAt, timezone)}`;
      row.append(name, times);
      timeList.append(row);
    }
    timeBox.hidden = false;
  } else {
    timeBox.hidden = true;
  }

  const plotted = segments.filter((segment) => segment.end > segment.start);
  if (!plotted.length) {
    panel.hidden = true;
    return;
  }
  const minTime = Math.min(...plotted.map((segment) => segment.start));
  const maxTime = Math.max(...plotted.map((segment) => segment.end));
  if (!(maxTime > minTime)) {
    panel.hidden = true;
    return;
  }
  const width = 600;
  const height = 220;
  const left = 45;
  const right = 12;
  const top = 20;
  const bottom = 40;
  const plotWidth = width - left - right;
  const plotHeight = height - top - bottom;
  const x = (time) => left + ((time - minTime) / (maxTime - minTime)) * plotWidth;
  const y = (level) => top + ((10 - (level == null ? -13 : Number(level))) / 24) * plotHeight;
  const svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
  svg.setAttribute("viewBox", `0 0 ${width} ${height}`);
  svg.setAttribute("role", "img");
  svg.setAttribute("aria-label", "Scheduled temperature target levels over time");
  svg.classList.add("timeline-svg");
  const make = (name, attrs) => {
    const element = document.createElementNS("http://www.w3.org/2000/svg", name);
    for (const [key, value] of Object.entries(attrs)) element.setAttribute(key, String(value));
    return element;
  };
  for (const level of [-10, 0, 10]) {
    const yy = y(level);
    svg.append(
      make("line", { x1: left, x2: width - right, y1: yy, y2: yy, class: "timeline-grid" }),
    );
    const label = make("text", {
      x: left - 10,
      y: yy + 4,
      "text-anchor": "end",
      class: "timeline-axis",
    });
    label.textContent = String(level);
    svg.append(label);
  }
  const offLabel = make("text", {
    x: left - 10,
    y: y(null) + 4,
    "text-anchor": "end",
    class: "timeline-axis timeline-off-label",
  });
  offLabel.textContent = "Off";
  svg.append(offLabel);
  const sorted = [...plotted].sort((a, b) => a.start - b.start);
  let previousLevel = null;
  let hasPrevious = false;
  sorted.forEach((segment) => {
    const x1 = x(segment.start);
    const x2 = x(segment.end);
    const yy = y(segment.level);
    const startY = hasPrevious ? y(previousLevel) : yy;
    const segmentPath = make("path", {
      d: `M ${x1} ${startY} V ${yy} H ${x2}`,
      class:
        segment.key === "off" || (segment.key === "padding" && segment.level == null)
          ? "timeline-line timeline-off-segment"
          : "timeline-line",
    });
    svg.append(segmentPath);
    previousLevel = segment.level;
    hasPrevious = true;
  });
  const startText = make("text", { x: left, y: height - 12, class: "timeline-axis" });
  startText.textContent = formatClock(minTime, timezone);
  const endText = make("text", {
    x: width - right,
    y: height - 12,
    "text-anchor": "end",
    class: "timeline-axis",
  });
  endText.textContent = formatClock(maxTime, timezone);
  svg.append(startText, endText);

  const now = Date.now();
  const active = sorted.find(
    (segment) =>
      !["off", "padding"].includes(segment.key) && now >= segment.start && now < segment.end,
  );
  if (active) {
    const nowX = x(now);
    svg.append(
      make("line", {
        x1: nowX,
        x2: nowX,
        y1: top,
        y2: top + plotHeight,
        class: "timeline-current",
      }),
    );
    const point = make("circle", { cx: nowX, cy: y(active.level), r: 5, class: "timeline-point" });
    svg.append(point);
    nowLabel.textContent = `Now · ${formatClock(now, timezone)} · scheduled level ${Number(active.level) > 0 ? "+" : ""}${active.level}`;
    nowLabel.hidden = false;
  }
  chart.append(svg);
  panel.hidden = false;
}

function setOptions(select, zones) {
  const list = zones?.length
    ? zones
    : Intl.supportedValuesOf?.("timeZone") || ["UTC", "America/Los_Angeles", "America/New_York"];
  select.replaceChildren(
    ...list.map((zone) => {
      const option = document.createElement("option");
      option.value = zone;
      option.textContent = zone.replaceAll("_", " ");
      return option;
    }),
  );
}

function applySettings(value) {
  settings = value;
  document.querySelector("#bedtime").value = value.bedtime || "22:00";
  document.querySelector("#wakeTime").value = value.wakeTime || "06:00";
  const timezone = document.querySelector("#timezone");
  if (![...timezone.options].some((option) => option.value === value.timezone)) {
    const option = document.createElement("option");
    option.value = value.timezone || "UTC";
    option.textContent = option.value;
    timezone.add(option);
  }
  timezone.value = value.timezone || "UTC";
  for (const name of levelNames) {
    const input = document.getElementById(name);
    const level = Math.max(-10, Math.min(10, Number(value.levels?.[name] ?? 0)));
    input.value = String(level);
    document.getElementById(`${name}-value`).value = String(level);
    document.getElementById(`${name}-value`).textContent = String(level);
  }
}

function renderStatus(status) {
  currentStatus = { ...currentStatus, ...status };
  renderTimeline(currentStatus);
  paused = Boolean(currentStatus.paused);
  const connected = Boolean(currentStatus.connected);
  const configured = Boolean(currentStatus.configured);
  const controlEnabled = Boolean(currentStatus.controlEnabled);
  const inPreview = connected && configured && !controlEnabled;
  document.querySelector("#status-value").textContent = !connected
    ? "Connect your account"
    : !configured
      ? "Save your schedule"
      : inPreview
        ? "Preview mode"
        : paused
          ? "Schedule paused"
          : "Schedule is on";
  const next = !configured
    ? "Choose your times and temperatures, then save"
    : inPreview
      ? "Schedule is not controlling the bed yet"
      : currentStatus.nextActionAt
        ? `Next · ${formatDate(currentStatus.nextActionAt)}`
        : currentStatus.nextActionLabel ||
          (connected ? "Waiting for next scheduled action" : "Connect Eight Sleep to get started");
  document.querySelector("#status-detail").textContent = next;
  document.querySelector("#account-detail").textContent = connected
    ? "Connected securely"
    : "Not connected yet";
  document.querySelector("#last-run").textContent =
    `Last run · ${formatDate(currentStatus.lastRunAt)}`;
  const icon = document.querySelector(".status-icon");
  icon.classList.toggle("paused", paused || inPreview);
  pauseButton.hidden = !connected || !configured || inPreview;
  pauseButton.textContent = paused ? "Resume" : "Pause";
  connectButton.textContent = connected ? "Reconnect" : "Connect";
  runError.textContent = currentStatus.lastError
    ? `Last run error: ${currentStatus.lastError}`
    : "";
  runError.hidden = !currentStatus.lastError;
  document.querySelector("#save-hint").textContent = controlEnabled
    ? "Saved changes are reconciled with the bed within about a minute."
    : "Schedule saved for preview. Bed control is currently off.";
}

async function load() {
  setOptions(document.querySelector("#timezone"));
  try {
    const [status, value] = await Promise.all([request("/api/status"), request("/api/settings")]);
    renderStatus(status);
    applySettings(value);
  } catch (error) {
    document.querySelector("#status-value").textContent = "Can’t load schedule";
    document.querySelector("#status-detail").textContent = "Check your connection and try again.";
    setNotice(error.message, "error");
  }
}

for (const button of document.querySelectorAll("[data-step]")) {
  button.addEventListener("click", () => {
    const input = document.getElementById(button.dataset.step);
    const next = Math.max(-10, Math.min(10, Number(input.value) + Number(button.dataset.delta)));
    input.value = String(next);
    document.getElementById(`${button.dataset.step}-value`).value = String(next);
    document.getElementById(`${button.dataset.step}-value`).textContent = String(next);
  });
}

form.addEventListener("submit", async (event) => {
  event.preventDefault();
  const data = new FormData(form);
  const payload = {
    bedtime: data.get("bedtime"),
    wakeTime: data.get("wakeTime"),
    timezone: data.get("timezone"),
    levels: Object.fromEntries(levelNames.map((name) => [name, Number(data.get(name))])),
  };
  saveButton.disabled = true;
  saveButton.querySelector("span").textContent = "Saving…";
  setNotice("");
  try {
    const result = await request("/api/settings", { method: "PUT", body: JSON.stringify(payload) });
    applySettings(result.settings || payload);
    renderStatus(await request("/api/status"));
    setNotice(
      currentStatus.controlEnabled
        ? "Schedule saved. Bed settings will be reconciled shortly."
        : "Schedule saved in preview mode. Bed control is off.",
    );
  } catch (error) {
    setNotice(error.message || "Could not save your schedule. Try again.", "error");
  } finally {
    saveButton.disabled = false;
    saveButton.querySelector("span").textContent = "Save schedule";
  }
});

pauseButton.addEventListener("click", async () => {
  pauseButton.disabled = true;
  try {
    const result = await request("/api/control", {
      method: "POST",
      body: JSON.stringify({ action: paused ? "resume" : "pause" }),
    });
    renderStatus({
      connected: true,
      paused: result.paused,
      controlEnabled: currentStatus.controlEnabled,
    });
    setNotice(result.paused ? "Schedule paused." : "Schedule resumed.");
  } catch (error) {
    setNotice(error.message || "Could not update schedule status.", "error");
  } finally {
    pauseButton.disabled = false;
  }
});

connectButton.addEventListener("click", () => {
  connectError.hidden = true;
  connectForm.reset();
  connectDialog.showModal();
});
document.querySelector("#close-dialog").addEventListener("click", () => connectDialog.close());
connectForm.addEventListener("submit", async (event) => {
  event.preventDefault();
  const submit = document.querySelector("#connect-submit");
  submit.disabled = true;
  submit.textContent = "Connecting…";
  connectError.hidden = true;
  try {
    await request("/api/connect", {
      method: "POST",
      body: JSON.stringify({
        email: document.querySelector("#email").value,
        password: document.querySelector("#password").value,
      }),
    });
    connectDialog.close();
    setNotice("Eight Sleep connected successfully.");
    await load();
  } catch (error) {
    connectError.textContent =
      error.message || "Could not connect. Check your details and try again.";
    connectError.hidden = false;
  } finally {
    submit.disabled = false;
    submit.textContent = "Connect securely";
  }
});

load();
window.setInterval(() => renderTimeline(currentStatus), 60_000);
